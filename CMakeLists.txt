cmake_minimum_required(VERSION 3.18)
project(NanoVLLMC LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(NANO_WITH_CUDA "Build C11 CUDA execution and operators" OFF)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_library(nano_model STATIC src/model.c src/cache.c src/tokenizer.c src/sampler.c)
target_include_directories(nano_model PUBLIC include)
target_link_libraries(nano_model PUBLIC m)
add_library(nano_cpu STATIC src/engine.c)
target_link_libraries(nano_cpu PUBLIC nano_model)
add_executable(nanovllm src/main.c)
target_link_libraries(nanovllm PRIVATE nano_cpu)
foreach(name engine tokenizer)
    add_executable(test_${name} tests/test_${name}.c)
    target_compile_options(test_${name} PRIVATE -UNDEBUG)
    target_link_libraries(test_${name} PRIVATE nano_cpu)
endforeach()
add_test(NAME fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures.py ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(fixtures PROPERTIES FIXTURES_SETUP data)
add_test(NAME engine_cpu COMMAND test_engine ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
add_test(NAME tokenizer COMMAND test_tokenizer ${CMAKE_CURRENT_BINARY_DIR}/fixtures/tokenizer.bin)
add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:nanovllm> ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(engine_cpu tokenizer cli PROPERTIES FIXTURES_REQUIRED data)
if(NANO_WITH_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_library(nano_operators STATIC src/operators/sgemv.cu src/operators/rmsnorm.cu src/operators/attention.cu src/operators/decode_ops.cu)
    target_include_directories(nano_operators PUBLIC include)
    target_link_libraries(nano_operators PUBLIC CUDA::cudart)
    set_target_properties(nano_operators PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(nano_operators PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
    add_library(nano_cuda STATIC src/cuda_runtime.c src/engine.c)
    target_compile_definitions(nano_cuda PRIVATE NANO_WITH_CUDA=1)
    target_link_libraries(nano_cuda PUBLIC nano_model nano_operators)
    add_executable(nanovllm_cuda src/main.c)
    set_target_properties(nanovllm_cuda PROPERTIES LINKER_LANGUAGE C)
    target_link_libraries(nanovllm_cuda PRIVATE nano_cuda)
    add_executable(bench_operators benchmarks/operators/bench_operators.c)
    set_target_properties(bench_operators PROPERTIES LINKER_LANGUAGE C)
    target_link_libraries(bench_operators PRIVATE nano_operators CUDA::cublas m)
    add_test(NAME gemv_rms COMMAND bench_operators --test)
    set_tests_properties(gemv_rms PROPERTIES SKIP_RETURN_CODE 77)
    add_executable(bench_decode benchmarks/round2/main.c benchmarks/round2/attention_bench.c benchmarks/round2/decode_bench.c)
    set_target_properties(bench_decode PROPERTIES LINKER_LANGUAGE C)
    target_link_libraries(bench_decode PRIVATE nano_cuda ${CMAKE_DL_LIBS})
    add_test(NAME attention COMMAND bench_decode --test-attention)
    add_test(NAME cuda_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cuda_fixtures.py ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
    set_tests_properties(cuda_fixtures PROPERTIES FIXTURES_SETUP cuda_data)
    add_test(NAME decode COMMAND bench_decode --test-decode ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
    set_tests_properties(attention decode PROPERTIES SKIP_RETURN_CODE 77)
    set_tests_properties(decode PROPERTIES FIXTURES_REQUIRED "data;cuda_data")
endif()
